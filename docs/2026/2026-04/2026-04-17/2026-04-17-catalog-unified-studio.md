# Catalog 통합뷰 구현 Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `/catalog`를 MasterDetail 통합뷰로 재구성 — 좌측 커버리지 매트릭스 + 우측 Storybook 상세. 컴포넌트 파일이 SSOT, 스캐너가 갭 데이터 파생.

**Architecture:** `scripts/scanCatalog.mjs`(ts-morph)가 `src/interactive-os/ui/**/*.tsx`를 파싱해 `catalog.generated.ts`로 `CatalogEntry[]` 방출. `PageCatalog.tsx`는 `ui/composites/MasterDetail` + `ui/Table` + `ui/composites/TabList` + `ui/CodeViewer`로 통합뷰 구성. demo 렌더는 기존 `demoRegistry.ts`의 `import.meta.glob` 재사용. 쓰기 액션은 Vite dev middleware(`src/server/catalogApi.ts`) 경유.

**Tech Stack:** React 19, Vite, ts-morph, `ax()`, `useAria`, Vitest.

**Spec:** `docs/superpowers/specs/2026-04-17-catalog-unified-studio-design.md`

```js
    return {
      name,
      path: relative(process.cwd(), src),
      demoPath: demos.has(demoPath) ? relative(process.cwd(), demoPath) : null,
      category: categorize(src),
    }
  })
}
```

```js
import { Project, SyntaxKind } from 'ts-morph'

const project = new Project({
  tsConfigFilePath: resolve('tsconfig.json'),
  skipAddingFilesFromTsConfig: true,
})

function analyzeSource(filePath) {
  const sf = project.addSourceFileAtPath(filePath)
  const result = {
    propsSignature: {},
    axAxes: [],
    ariaUsed: false,
    jsDoc: null,
  }

  // 1) ARIA: useAria / AriaComponentProps import 존재
  const imports = sf.getImportDeclarations().flatMap((d) => d.getNamedImports().map((n) => n.getName()))
  result.ariaUsed = imports.includes('useAria') || imports.includes('AriaComponentProps')

  // 2) props 시그니처: 주 export의 props 타입 추출
  const main = sf.getInterfaces().find((i) => /Props$/.test(i.getName()))
    || sf.getTypeAliases().find((t) => /Props$/.test(t.getName()))
  if (main) {
    const members = main.getKind() === SyntaxKind.InterfaceDeclaration
      ? main.getProperties()
      : []  // 타입 alias의 union prop은 복잡 — 일단 interface만
    for (const prop of members) {
      const tn = prop.getTypeNode()?.getText() ?? ''
      const literals = [...tn.matchAll(/'([^']+)'/g)].map((m) => m[1])
      if (literals.length > 0) result.propsSignature[prop.getName()] = literals
    }
  }

  // 3) ax 축: ax({...}) 호출 수집
  const calls = sf.getDescendantsOfKind(SyntaxKind.CallExpression)
  for (const c of calls) {
    if (c.getExpression().getText() !== 'ax') continue
    const arg = c.getArguments()[0]
    if (arg?.getKind() === SyntaxKind.ObjectLiteralExpression) {
      for (const p of arg.getProperties()) {
        const name = p.getName?.()
        if (name) result.axAxes.push(name)
      }
    }
  }
  result.axAxes = [...new Set(result.axAxes)]

  // 4) JSDoc: 주 export function/const 위의 JSDoc
  const main2 = sf.getFunctions().find((f) => f.isExported())
    || sf.getVariableStatements().find((v) => v.isExported())
  const jsDocs = main2?.getJsDocs?.()
  if (jsDocs && jsDocs.length > 0) result.jsDoc = jsDocs[0].getDescription().trim()

  project.removeSourceFile(sf)
  return result
}
```

```js
function analyzeDemo(demoPath, componentName, signature) {
  const sf = project.addSourceFileAtPath(demoPath)
  const jsx = sf.getDescendantsOfKind(SyntaxKind.JsxOpeningElement)
    .concat(sf.getDescendantsOfKind(SyntaxKind.JsxSelfClosingElement))
  const combos = []
  for (const el of jsx) {
    if (el.getTagNameNode().getText() !== componentName) continue
    const combo = {}
    for (const attr of el.getAttributes()) {
      if (attr.getKind() !== SyntaxKind.JsxAttribute) continue
      const name = attr.getNameNode().getText()
      const init = attr.getInitializer()
      if (!init) { combo[name] = 'true'; continue }
      const txt = init.getText().replace(/^['"{]|['"}]$/g, '')
      combo[name] = txt
    }
    combos.push(combo)
  }
  project.removeSourceFile(sf)

  // 시그니처 기반 예상 조합 (1-axis가 아닌 cartesian은 과함 — 축별 리터럴 최소 1회 사용만 검증)
  const missing = []
  let total = 0, covered = 0
  for (const [prop, literals] of Object.entries(signature)) {
    for (const lit of literals) {
      total++
      const hit = combos.some((c) => c[prop] === lit)
      if (hit) covered++
      else missing.push(`${prop}=${lit}`)
    }
  }
  return { covered, total, missing }
}
```

```ts
// src/pages/catalog/catalogTransform.ts
import type { CatalogEntry, GapStatus } from './catalogTypes'

export interface MatrixRow {
  id: string           // entry.name
  name: string
  category: string
  cells: {
    demo: GapStatus
    coverage: GapStatus
    ax: GapStatus
    aria: GapStatus
    docs: GapStatus
  }
  hasAnyGap: boolean
}

function coverageStatus(e: CatalogEntry): GapStatus {
  const { covered, total } = e.gaps.branchCoverage
  if (total === 0) return 'ok'
  if (covered === total) return 'ok'
  if (covered === 0) return 'missing'
  return 'partial'
}

function axStatus(e: CatalogEntry): GapStatus {
  return e.gaps.axAxes.missing.length === 0 ? 'ok' : 'partial'
}

export function toMatrixRows(entries: CatalogEntry[]): MatrixRow[] {
  return entries.map((e) => {
    const cells = {
      demo: e.gaps.demoFile === 'ok' ? 'ok' as const : 'missing' as const,
      coverage: coverageStatus(e),
      ax: axStatus(e),
      aria: e.gaps.aria,
      docs: e.gaps.docs,
    }
    const hasAnyGap = Object.values(cells).some((s) => s !== 'ok')
    return { id: e.name, name: e.name, category: e.category, cells, hasAnyGap }
  })
}

export function summarize(rows: MatrixRow[]) {
  return {
    total: rows.length,
    withGaps: rows.filter((r) => r.hasAnyGap).length,
    byAxis: {
      demo: rows.filter((r) => r.cells.demo !== 'ok').length,
      coverage: rows.filter((r) => r.cells.coverage !== 'ok').length,
      ax: rows.filter((r) => r.cells.ax !== 'ok').length,
      aria: rows.filter((r) => r.cells.aria !== 'ok').length,
      docs: rows.filter((r) => r.cells.docs !== 'ok').length,
    },
  }
}
```

```ts
// src/server/catalogApi.ts
import { readFileSync, writeFileSync, existsSync } from 'node:fs'
import { resolve } from 'node:path'
import { Project, SyntaxKind } from 'ts-morph'

export function catalogApiPlugin() {
  return {
    name: 'catalog-api',
    configureServer(server: any) {
      server.middlewares.use('/api/catalog/scaffold-demo', async (req: any, res: any) => {
        if (req.method !== 'POST') return res.end()
        const body = await readJson(req)
        const demoPath = body.path.replace(/\.tsx$/, '.demo.tsx')
        if (existsSync(demoPath)) return reply(res, 409, { error: 'exists' })
        const content = `import { ${body.name} } from './${body.name}'\n\n` +
          `export const meta = { slug: '${body.name.toLowerCase()}', label: '${body.name}' }\n` +
          `export function Demo() {\n  return <${body.name} />\n}\n`
        writeFileSync(demoPath, content, 'utf8')
        reply(res, 200, { demoPath })
      })

      server.middlewares.use('/api/catalog/append-branch', async (req: any, res: any) => {
        if (req.method !== 'POST') return res.end()
        const body = await readJson(req)
        const [prop, value] = body.combo.split('=')
        const demoPath = resolve(body.demoPath)
        const src = readFileSync(demoPath, 'utf8')
        // 단순 append: Demo 리턴 JSX 바로 앞에 <Component {prop}="{value}" /> 삽입 시도
        const injected = src.replace(
          /(return\s*\()/,
          `return (\n      <${body.name} ${prop}="${value}" />\n      /* inserted */\n      `,
        )
        writeFileSync(demoPath, injected, 'utf8')
        reply(res, 200, { ok: true })
      })
    },
  }
}

function readJson(req: any): Promise<any> {
  return new Promise((r) => {
    const chunks: Buffer[] = []
    req.on('data', (c: Buffer) => chunks.push(c))
    req.on('end', () => { try { r(JSON.parse(Buffer.concat(chunks).toString())) } catch { r({}) } })
  })
}
function reply(res: any, code: number, data: any) {
  res.statusCode = code
  res.setHeader('content-type', 'application/json')
  res.end(JSON.stringify(data))
}
```
